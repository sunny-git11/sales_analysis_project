# Product list and price analysis
products = {
    "Laptop": 1200,
    "Smartphone": 800,
    "Headphones": 150,
    "Smartwatch": 250,
    "Tablet": 600,
    "Gaming PC": 2000  # Premium product
}

highest_priced_product = max(products, key=products.get)
lowest_priced_product = min(products, key=products.get)

print("Highest priced product: {} - ${}".format(highest_priced_product, products[highest_priced_product]))
print("Lowest priced product: {} - ${}".format(lowest_priced_product, products[lowest_priced_product]))

# Mid-range product price analysis

highest_mid_priced_product = max(mid_range_products, key=mid_range_products.get)
lowest_mid_priced_product = min(mid_range_products, key=mid_range_products.get)

print("Highest priced mid-range product: {} - ${}".format(highest_mid_priced_product, mid_range_products[highest_mid_priced_product]))
print("Lowest priced mid-range product: {} - ${}".format(lowest_mid_priced_product, mid_range_products[lowest_mid_priced_product]))

# Function to take stock level as input and print status
def check_stock_level():
    stock_level = int(input("Enter stock level: "))
    threshold = 20  # Define reorder threshold
    if stock_level > threshold:
        print("Stock is sufficient.")
    else:
        print("Reorder Now!")

# Function to calculate area of store section
def calculate_area(length, width):
    return length * width

# Example store sections and area calculation
sections = {
    "Electronics": (30, 20),
    "Clothing": (25, 15),
    "Groceries": (40, 35)
}

print("\nStore Section Areas:")
for section, dimensions in sections.items():
    area = calculate_area(dimensions[0], dimensions[1])
    print("{} section area: {} square meters".format(section, area))

# Loyalty points calculation
def calculate_loyalty_points(purchase_amount):
    if purchase_amount >= 1000:
        return purchase_amount * 2  # High spender multiplier
    elif purchase_amount >= 500:
        return purchase_amount * 1.5  # Medium spender multiplier
    else:
        return purchase_amount * 1  # Low spender multiplier

customers_spending = {
    "Alice": 1200,
    "Bob": 750,
    "Charlie": 300,
    "David": 1800,
    "Eve": 500
}

print("\nLoyalty Points Earned:")
for customer, amount in customers_spending.items():
    points = calculate_loyalty_points(amount)
    print("{} earned {} points".format(customer, points))
